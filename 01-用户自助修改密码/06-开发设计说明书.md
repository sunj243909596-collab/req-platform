```ts
export class UserError extends Error {
  constructor(
    public readonly code: string,
    message: string,
    public readonly http: number = 400,
    public readonly field?: string,
  ) {
    super(message);
    this.name = 'UserError';

export const UserErr = {

  sameAsOld:   () => new UserError('PASSWORD_UNCHANGED',   '新密码不能与旧密码相同', 400, 'newPassword'),
  confirmMismatch: () => new UserError('CONFIRM_MISMATCH', '两次输入的新密码不一致', 400, 'confirmPassword'),
  missingField: (n: string) => new UserError('MISSING_FIELD', `缺少必填字段: ${n}`, 400, n),
```

### 3.2 `server/src/services/user.service.ts`（在现有 110 行文件后追加）

```ts
import { UserError, UserErr } from './user-error';
import { invalidatePermissions } from './permission-cache.service';
import { writeActivityLog } from '../utils/activity-log';

export interface ChangeOwnPasswordInput {
  userId: number;
  username: string;            // 用于 lastPasswordChangeActor
  currentPassword: string;
  newPassword: string;
  confirmPassword: string;
}

export async function changeMyPassword(input: ChangeOwnPasswordInput) {
  if (!input.currentPassword) throw UserErr.missingField('currentPassword');
  if (!input.newPassword)     throw UserErr.missingField('newPassword');
  if (!input.confirmPassword) throw UserErr.missingField('confirmPassword');

  if (input.newPassword !== input.confirmPassword) throw UserErr.confirmMismatch();

  const user = await prisma.user.findUnique({
    where: { id: input.userId },
    select: { passwordHash: true, isActive: true },
  });
  if (!user) throw new UserError('USER_NOT_FOUND', '用户不存在', 404);
  if (!user.isActive) throw new UserError('USER_DISABLED', '账户已停用', 403);

  const bcryptLib = (bcrypt as { default?: typeof bcrypt }).default || bcrypt;
  const ok = await bcryptLib.compare(input.currentPassword, user.passwordHash);
  if (!ok) {
    await writeActivityLog({
      actorUserId: input.userId,
      eventType: 'PASSWORD_CHANGE_REJECTED',
      payload: { reason: 'OLD_MISMATCH' },
    });
    throw UserErr.oldMismatch();
  }

  try {
    validatePassword(input.newPassword);
  } catch (e) {
    await writeActivityLog({
      actorUserId: input.userId,
      eventType: 'PASSWORD_CHANGE_REJECTED',
      payload: { reason: 'WEAK' },
    });
    throw new UserError('WEAK_PASSWORD', (e as Error).message, 400, 'newPassword');
  }

  if (await bcryptLib.compare(input.newPassword, user.passwordHash)) {
    await writeActivityLog({
      actorUserId: input.userId,
      eventType: 'PASSWORD_CHANGE_REJECTED',
      payload: { reason: 'SAME_AS_OLD' },
    });
    throw UserErr.sameAsOld();
  }

  const newHash = await bcryptLib.hash(input.newPassword, 10);
  const updated = await prisma.$transaction(async (tx) => {
    const u = await tx.user.update({
      where: { id: input.userId },
      data: {
        passwordHash: newHash,
        tokenVersion: { increment: 1 },
        passwordChangedAt: new Date(),
        lastPasswordChangeActor: input.username,
      },
      select: { passwordChangedAt: true },
    });
    await writeActivityLog({
      actorUserId: input.userId,
      eventType: 'USER_CHANGE_PASSWORD',
      payload: { source: 'self' },
      tx,
    });
    return u;
  });

  invalidatePermissions(input.userId);
  return { passwordChangedAt: updated.passwordChangedAt };
}

export interface ResetPasswordByAdminInput {
  targetUserId: number;
  adminUsername: string;
  newPassword: string;
}

export async function resetPasswordByAdmin(input: ResetPasswordByAdminInput) {
  const target = await prisma.user.findUnique({
    where: { id: input.targetUserId },
    select: { username: true },
  });
  if (!target) throw new UserError('USER_NOT_FOUND', '用户不存在', 404);

  validatePassword(input.newPassword);
  const bcryptLib = (bcrypt as { default?: typeof bcrypt }).default || bcrypt;
  const newHash = await bcryptLib.hash(input.newPassword, 10);

  await prisma.$transaction(async (tx) => {
    await tx.user.update({
      where: { id: input.targetUserId },
      data: {
        passwordHash: newHash,
        tokenVersion: { increment: 1 },
        passwordChangedAt: new Date(),
        lastPasswordChangeActor: `admin:${input.adminUsername}`,
        mustChangePassword: true,
      },
    });
    await writeActivityLog({
      actorUserId: input.targetUserId,
      eventType: 'ADMIN_RESET_PASSWORD',
      payload: {
        targetUserId: input.targetUserId,
        targetUsername: target.username,
        adminUsername: input.adminUsername,
      },
      tx,
    });
  });

  invalidatePermissions(input.targetUserId);
}
```

> **要点**：
> - `bcrypt as { default?: typeof bcrypt }` 写法延续现有 70 行风格
> - `writeActivityLog` 同事务写入，避免审计缺口
> - `tokenVersion: { increment: 1 }` Prisma 原生操作

### 3.3 `server/src/routes/me.routes.ts`（新建）

```ts
import { Hono } from 'hono';
import { authMiddleware } from '../middleware/auth';
import { changePasswordLimiter } from '../middleware/rate-limit';
import { changeMyPassword } from '../services/user.service';
import { UserError } from '../services/user-error';

export function createMeRoutes() {
  const app = new Hono();

  app.post('/password', authMiddleware, changePasswordLimiter, async (c) => {
    const body = (await c.req.json().catch(() => ({}))) as Record<string, string>;
    try {
      const result = await changeMyPassword({
        userId: c.get('userId') as number,
        username: c.get('username') as string,
        currentPassword: body.currentPassword ?? '',
        newPassword: body.newPassword ?? '',
        confirmPassword: body.confirmPassword ?? '',
      });
      c.header('Cache-Control', 'no-store');
      return c.json({
        ok: true,
        passwordChangedAt: result.passwordChangedAt!.toISOString(),
        message: '密码已修改，请重新登录',
      });
    } catch (e) {
      if (e instanceof UserError) {
        return c.json(
          { error: e.message, errorCode: e.code, field: e.field },
          e.http as 400 | 401 | 403 | 404,
        );
      }
      // 兜底
      return c.json({ error: '内部错误', errorCode: 'INTERNAL' }, 500);
    }
  });

  return app;
}
```

### 3.5 `server/src/middleware/rate-limit.ts`（追加）

```ts
export const changePasswordLimiter: MiddlewareHandler = async (c, next) => {
  const userId = c.get('userId');
  const key = `pwd:${userId}`;
  const { allowed, retryAfterSec } = await consumeBucket(key, {
    capacity: 3,                 // 3 次
    refillPerSec: 1 / 60,        // 1 次/分钟
  });
  if (!allowed) {
    return c.json(
      { error: `操作过于频繁，请 ${retryAfterSec} 秒后重试`, errorCode: 'RATE_LIMITED' },
      429,
    );
  }
  await next();
};
```

> 复用现有 `consumeBucket` 风格（项目内已存在）；如不存在则用 `lru-cache` 自实现。

### 3.6 `server/src/utils/activity-log.ts`（新建）

```ts
import type { Prisma, PrismaClient } from '@prisma/client';
import { prisma } from '../lib/prisma';

type ActorUserId = number;

export async function writeActivityLog(args: {
  actorUserId: ActorUserId;
  eventType: 'USER_CHANGE_PASSWORD' | 'ADMIN_RESET_PASSWORD' | 'PASSWORD_CHANGE_REJECTED';
  payload: Record<string, unknown>;
  tx?: Prisma.TransactionClient;
}) {
  const client = (args.tx ?? prisma) as PrismaClient;
  await client.activityLog.create({
    data: {
      actorUserId: args.actorUserId,
      eventType: args.eventType,
      payload: args.payload as any,
      createdAt: new Date(),
    },
  });
}
```

### 4.3 `client/src/app/components/ChangePasswordDialog.tsx`（新建）

骨架：

```tsx
import { useState } from 'react';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { toast } from 'sonner';
import { Eye, EyeOff, Loader2, KeyRound } from 'lucide-react';
import { changeMyPassword } from '@/api/users';
import { useNavigate } from 'react-router';

interface Props { open: boolean; onOpenChange: (b: boolean) => void; }

function strengthScore(p: string): number {
  let n = 0;
  if (p.length >= 8) n++;
  if (/[A-Z]/.test(p)) n++;
  if (/[a-z]/.test(p)) n++;
  if (/[0-9]/.test(p)) n++;
  if (/[^A-Za-z0-9]/.test(p)) n++;
  return n;       // 0~5
}

export function ChangePasswordDialog({ open, onOpenChange }: Props) {
  const [cur, setCur] = useState('');
  const [np, setNp]   = useState('');
  const [cf, setCf]   = useState('');
  const [show, setShow] = useState(false);
  const [busy, setBusy] = useState(false);
  const [errs, setErrs] = useState<{ current?: string; next?: string; confirm?: string }>({});
  const nav = useNavigate();

  const score = strengthScore(np);

  const submit = async () => {
    setErrs({});
    if (np !== cf) { setErrs({ confirm: '两次输入的新密码不一致' }); return; }
    if (np === cur) { setErrs({ next: '新密码不能与当前密码相同' }); return; }
    if (score < 4) { setErrs({ next: '密码需 ≥ 8 位且包含大小写字母、数字' }); return; }
    setBusy(true);
    try {
      await changeMyPassword({ currentPassword: cur, newPassword: np, confirmPassword: cf });
      toast.success('密码已修改，请重新登录');
      setBusy(false);
      onOpenChange(false);
      setTimeout(() => {
        localStorage.removeItem('authToken');
        nav('/login');
      }, 1500);
    } catch (e: any) {
      const code = e?.response?.data?.errorCode;
      const field = e?.response?.data?.field;
      const msg   = e?.response?.data?.error ?? '修改失败';
      if (field === 'currentPassword') setErrs({ current: msg });
      else if (field === 'newPassword') setErrs({ next: msg });
      else if (field === 'confirmPassword') setErrs({ confirm: msg });
      else toast.error(msg);
      setBusy(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader><DialogTitle><KeyRound className="inline mr-2" /> 修改密码</DialogTitle></DialogHeader>

        <div className="space-y-4 mt-2">
          <FormField label="当前密码" error={errs.current}>
            <Input type={show ? 'text' : 'password'} value={cur} onChange={(e) => setCur(e.target.value)} autoFocus />
          </FormField>

          <FormField label="新密码" error={errs.next} hint="≥ 8 位，必须包含大小写字母与数字">
            <Input type={show ? 'text' : 'password'} value={np} onChange={(e) => setNp(e.target.value)} />
            <StrengthBar n={score} />
          </FormField>

          <FormField label="确认新密码" error={errs.confirm}>
            <Input type={show ? 'text' : 'password'} value={cf} onChange={(e) => setCf(e.target.value)} />
          </FormField>

          <div className="flex items-center gap-2 text-sm">
            <input id="show" type="checkbox" checked={show} onChange={() => setShow(v => !v)} />
            <label htmlFor="show">显示密码</label>
          </div>

          <div className="flex justify-end gap-2 pt-2">
            <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={busy}>取消</Button>
            <Button onClick={submit} disabled={busy}>
              {busy && <Loader2 className="mr-2 h-4 w-4 animate-spin" />} 保存修改
            </Button>
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}
```

> `FormField` / `StrengthBar` 为同文件内的两个小内部组件，按项目风格写。

## 8. 风险与回滚

| 风险 | 触发 | 应急 |
|---|---|---|
| `authMiddleware` 引入 DB 查询 → 性能回退 | 命中率低 | 加 Redis 缓存 `tokenVersion`，TTL=60s |
| 旧 token 401 风暴 | 改密瞬间其它窗口 | 客户端拦截器 idempotent；服务端限流兜底 |
| 误提交 `mustChangePassword` | ADMIN 重置自己 | 自助改密将 `mustChangePassword=false` |

> **紧急回滚**：路由层一行 `app.post('/password', (c) => c.json({ error: 'maintenance' }, 503))` 即可止血；DB 列可保留。
